import cv2

cap = cv2.VideoCapture(2)#이부분만 수정하면됨

while True:
    ret, frame = cap.read()
    if not ret:
        break

    cv2.imshow("cam", frame)

    if cv2.waitKey(1) == 27:
        break

cap.release()
cv2.destroyAllWindows()
EOF
